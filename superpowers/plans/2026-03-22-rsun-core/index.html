rsun-core: CPU Reference Solar Radiation Library — Implementation Plan¶

For agentic workers: REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (- [ ]) syntax for tracking.

Goal: Port the GRASS GIS r.sun solar radiation algorithm (~2,800 lines of C) to a pure Rust library with CPU parallelism, validated against GRASS r.sun output on sol/examples/mcn_10m.tif.

Architecture: A Rust workspace with rsun-core as a zero-dependency library crate containing solar math (solar.rs), radiation calculations (radiation.rs), horizon angles (horizon.rs), terrain derivatives (terrain.rs), and GeoTIFF I/O (io.rs). A simple rsun-cli binary wraps the library for command-line use. All math ported from GRASS r.sun's main.c (2,146 lines) and rsunlib.c (678 lines).

Tech Stack: Rust 1.75+, gdal crate (GeoTIFF I/O), proj crate (CRS transforms), rayon (CPU parallelism), clap 4.x (CLI), approx (float comparison in tests)

Spec: docs/superpowers/specs/2026-03-22-rsun-gpu-design.md

GRASS r.sun source reference: https://github.com/OSGeo/grass/tree/main/raster/r.sun

This is Phase 1 of 4. Phases 2-4 (GPU shaders, Python bindings, WebAssembly) will have separate plans.

/// A 2D grid of f32 values with geographic metadata
#[derive(Debug, Clone)]
pub struct Grid {
    pub data: Vec<f32>,
    pub rows: usize,
    pub cols: usize,
    pub nodata: f32,
}

impl Grid {
    pub fn new(rows: usize, cols: usize, nodata: f32) -> Self {
        Self {
            data: vec![nodata; rows * cols],
            rows,
            cols,
            nodata,
        }
    }

    pub fn get(&self, row: usize, col: usize) -> f32 {
        self.data[row * self.cols + col]
    }

    pub fn set(&mut self, row: usize, col: usize, value: f32) {
        self.data[row * self.cols + col] = value;
    }

    pub fn is_nodata(&self, row: usize, col: usize) -> bool {
        let v = self.get(row, col);
        v.is_nan() || v == self.nodata
    }
}

// rsun/rsun-core/src/solar.rs
//! Solar position calculations.
//!
//! Ported from GRASS GIS r.sun rsunlib.c (Hofierka & Suri 2002).
//! All angles in radians unless noted otherwise.

use std::f64::consts::PI;

const PI2: f64 = 2.0 * PI;
const HOURANGLE: f64 = PI / 12.0; // 15 degrees per hour in radians

/// Solar declination for a given day of year.
///
/// From GRASS `com_declin()`. Returns declination in radians.
/// Positive in northern summer, negative in northern winter.
pub fn declination(day: u16) -> f64 {
    let d1 = PI2 * (day as f64) / 365.25;
    asin_safe(0.3978 * (d1 - 1.4 + 0.0355 * (d1 - 0.0489).sin()).sin())
}

/// Solar constant corrected for Earth-Sun distance on given day.
///
/// From GRASS `com_sol_const()`. Earth is closest to Sun (~Jan 3)
/// and furthest (~Jul 6). Orbital eccentricity = 0.01672.
pub fn corrected_solar_constant(day: u16, solar_constant: f64) -> f64 {
    let d1 = PI2 * (day as f64) / 365.25;
    solar_constant * (1.0 + 0.03344 * (d1 - 0.048869).cos())
}

/// Compute sunrise and sunset times (in decimal hours, solar time).
///
/// From GRASS `com_par_const()` sunrise/sunset logic.
/// Returns (sunrise_hour, sunset_hour).
pub fn sunrise_sunset(latitude: f64, declination: f64) -> (f64, f64) {
    let sin_lat = latitude.sin();
    let cos_lat = latitude.cos();
    let cos_decl = declination.cos();
    let sin_decl = declination.sin();

    let lum_c31 = cos_lat * cos_decl;
    let lum_c33 = sin_lat * sin_decl;

    if lum_c31.abs() < 1e-4 {
        if lum_c33 > 0.0 {
            return (0.0, 24.0); // sun never sets
        } else {
            return (12.0, 12.0); // sun never rises
        }
    }

    let pom = -lum_c33 / lum_c31;
    if pom.abs() <= 1.0 {
        let pom_deg = pom.acos().to_degrees();
        let sunrise = (90.0 - pom_deg) / 15.0 + 6.0;
        let sunset = (pom_deg - 90.0) / 15.0 + 18.0;
        (sunrise, sunset)
    } else if pom < 0.0 {
        (0.0, 24.0) // sun above horizon all day
    } else {
        (12.0, 12.0) // sun below horizon all day
    }
}

/// Compute solar altitude and azimuth for a given time angle.
///
/// From GRASS `com_par()`. Time angle is in radians:
/// 0 = solar noon, negative = morning, positive = afternoon.
/// HOURANGLE = pi/12 radians per hour.
///
/// Returns (solar_altitude, solar_azimuth) in radians.
pub fn solar_position(latitude: f64, declination: f64, time_angle: f64) -> (f64, f64) {
    let sin_lat = latitude.sin();
    let cos_lat = latitude.cos();
    let cos_decl = declination.cos();
    let sin_decl = declination.sin();

    let lum_c11 = sin_lat * cos_decl;
    let lum_c13 = -cos_lat * sin_decl;
    let lum_c22 = cos_decl;
    let lum_c31 = cos_lat * cos_decl;
    let lum_c33 = sin_lat * sin_decl;

    let cos_time = time_angle.cos();
    let sin_time = time_angle.sin();

    let lum_lx = -lum_c22 * sin_time;
    let lum_ly = lum_c11 * cos_time + lum_c13;
    let sin_solar_alt = lum_c31 * cos_time + lum_c33;

    let solar_altitude = sin_solar_alt.asin();

    let pom = (lum_lx * lum_lx + lum_ly * lum_ly).sqrt();
    let solar_azimuth = if pom > 1e-4 {
        let mut az = (lum_ly / pom).acos();
        if lum_lx < 0.0 {
            az = PI2 - az;
        }
        az
    } else {
        0.0 // undefined at zenith
    };

    (solar_altitude, solar_azimuth)
}

/// Time angle from decimal hour (solar time).
/// Noon = 0, morning = negative, afternoon = positive.
pub fn hour_to_time_angle(hour: f64) -> f64 {
    (hour - 12.0) * HOURANGLE
}

/// HOURANGLE constant: pi/12 radians per hour (15 degrees).
pub fn hourangle() -> f64 {
    HOURANGLE
}

/// Safe asin that clamps input to [-1, 1].
fn asin_safe(x: f64) -> f64 {
    x.clamp(-1.0, 1.0).asin()
}

// rsun/rsun-core/src/radiation.rs
//! Solar radiation components: beam (direct), diffuse, and reflected.
//!
//! Ported from GRASS GIS r.sun rsunlib.c brad() and drad() functions.
//! Implements the ESRA clear-sky radiation model.

use std::f64::consts::PI;

const RAD2DEG: f64 = 180.0 / PI;
const UNDEF: f64 = f64::MAX;
const EARTHRADIUS: f64 = 6371000.0;

/// Beam (direct) radiation on a tilted surface.
///
/// Port of GRASS `brad()`. Returns (beam_tilted, beam_horizontal).
///
/// # Arguments
/// * `s0` - cos(incidence angle) on tilted surface. If <= 0, surface faces away from sun.
/// * `solar_alt` - Solar altitude angle [radians]
/// * `elevation` - Ground elevation [meters]
/// * `linke` - Linke turbidity factor
/// * `cbh` - Real-sky beam coefficient (1.0 for clear sky)
/// * `g_norm_extra` - Corrected extraterrestrial irradiance [W/m²]
pub fn brad(
    s0: f64,
    solar_alt: f64,
    elevation: f64,
    linke: f64,
    cbh: f64,
    g_norm_extra: f64,
) -> (f64, f64) {
    if s0 <= 0.0 || solar_alt <= 0.0 {
        return (0.0, 0.0);
    }

    let sin_solar_alt = solar_alt.sin();

    // Atmospheric refraction correction
    let temp1 = 0.1594 + solar_alt * (1.123 + 0.065656 * solar_alt);
    let temp2 = 1.0 + solar_alt * (28.9344 + 277.3971 * solar_alt);
    let drefract = 0.061359 * temp1 / temp2;
    let h0refract = solar_alt + drefract;

    // Elevation correction for atmospheric pressure
    let elevation_corr = (-elevation / 8434.5_f64).exp();

    // Optical air mass (Kasten & Young 1989)
    let optical_air_mass = elevation_corr
        / (h0refract.sin()
            + 0.50572 * (h0refract * RAD2DEG + 6.07995).powf(-1.6364));

    // Rayleigh optical thickness
    let air_mass_2_linke = 0.8662 * linke;
    let rayl = if optical_air_mass <= 20.0 {
        1.0 / (6.6296
            + optical_air_mass
                * (1.7513
                    + optical_air_mass
                        * (-0.1202
                            + optical_air_mass * (0.0065 - optical_air_mass * 0.00013))))
    } else {
        1.0 / (10.4 + 0.718 * optical_air_mass)
    };

    // Horizontal beam irradiance
    let beam_h = cbh * g_norm_extra * sin_solar_alt
        * (-rayl * optical_air_mass * air_mass_2_linke).exp();

    // Beam on tilted surface
    let beam = beam_h * s0 / sin_solar_alt;

    (beam, beam_h)
}

/// Diffuse and reflected radiation on a tilted surface.
///
/// Port of GRASS `drad()`. Returns (diffuse, reflected).
///
/// # Arguments
/// * `s0` - cos(incidence angle) on tilted surface
/// * `beam_h` - Horizontal beam irradiance from `brad()`
/// * `solar_alt` - Solar altitude [radians]
/// * `solar_azimuth` - Solar azimuth [radians]
/// * `slope` - Surface slope [radians]
/// * `aspect` - Surface aspect [radians]
/// * `linke` - Linke turbidity factor
/// * `albedo` - Ground albedo
/// * `cdh` - Real-sky diffuse coefficient (1.0 for clear sky)
/// * `g_norm_extra` - Corrected extraterrestrial irradiance
pub fn drad(
    s0: f64,
    beam_h: f64,
    solar_alt: f64,
    solar_azimuth: f64,
    slope: f64,
    aspect: f64,
    linke: f64,
    albedo: f64,
    cdh: f64,
    g_norm_extra: f64,
) -> (f64, f64) {
    let sin_solar_alt = solar_alt.sin();
    let cos_slope = slope.cos();
    let sin_slope = slope.sin();

    // Diffuse transmission function (Suri model)
    let tn = -0.015843 + linke * (0.030543 + 0.0003797 * linke);
    let a1b = 0.26463 + linke * (-0.061581 + 0.0031408 * linke);
    let a1 = if a1b * tn < 0.0022 {
        0.0022 / tn
    } else {
        a1b
    };
    let a2 = 2.04020 + linke * (0.018945 - 0.011161 * linke);
    let a3 = -1.3025 + linke * (0.039231 + 0.0085079 * linke);

    let fd = a1 + a2 * sin_solar_alt + a3 * sin_solar_alt * sin_solar_alt;
    let dh = cdh * g_norm_extra * fd * tn; // horizontal diffuse
    let gh = beam_h + dh; // global horizontal

    // Flat surface: no geometric correction needed
    if slope.abs() < 1e-6 {
        return (dh, 0.0);
    }

    // Tilted surface corrections
    let kb = beam_h / (g_norm_extra * sin_solar_alt);
    let r_sky = (1.0 + cos_slope) / 2.0;

    let mut a_ln = solar_azimuth - aspect;
    if a_ln > PI {
        a_ln -= 2.0 * PI;
    } else if a_ln < -PI {
        a_ln += 2.0 * PI;
    }

    let fg = sin_slope - slope * cos_slope
        - PI * (0.5 * slope).sin() * (0.5 * slope).sin();

    let fx = if s0 <= 0.0 {
        // In shadow
        r_sky + fg * 0.252271
    } else if solar_alt >= 0.1 {
        ((0.00263 - kb * (0.712 + 0.6883 * kb)) * fg + r_sky) * (1.0 - kb)
            + kb * s0 / sin_solar_alt
    } else {
        // Low sun
        ((0.00263 - 0.712 * kb - 0.6883 * kb * kb) * fg + r_sky) * (1.0 - kb)
            + kb * sin_slope * a_ln.cos() / (0.1 - 0.008 * solar_alt)
    };

    let diffuse = dh * fx;
    let reflected = albedo * gh * (1.0 - cos_slope) / 2.0;

    (diffuse, reflected)
}

/// Cosine of the angle of incidence on a tilted surface.
///
/// From GRASS `lumcline2()` / Jenco formula. Returns the "s0" value
/// used by brad() and drad(). When s0 <= 0, the surface faces away from the sun.
///
/// # Arguments
/// * `slope` - Surface slope [radians]
/// * `aspect` - Surface aspect [radians, 0=east counterclockwise in GRASS convention]
/// * `latitude` - Geographic latitude [radians]
/// * `declination` - Solar declination [radians]
/// * `time_angle` - Hour angle [radians, 0 = noon]
pub fn cos_incidence(
    slope: f64,
    aspect: f64,
    latitude: f64,
    declination: f64,
    time_angle: f64,
) -> f64 {
    let sin_lat = latitude.sin();
    let cos_lat = latitude.cos();
    let cos_decl = declination.cos();
    let sin_decl = declination.sin();

    // Equivalent slope geometry (Jenco transformation)
    let cos_u = (PI / 2.0 - slope).cos(); // = sin(slope)
    let sin_u = (PI / 2.0 - slope).sin(); // = cos(slope)
    let cos_v = (PI / 2.0 + aspect).cos();
    let sin_v = (PI / 2.0 + aspect).sin();

    let sin_phi_l = -cos_lat * cos_u * sin_v + sin_lat * sin_u;
    let q1 = sin_lat * cos_u * sin_v + cos_lat * sin_u;

    let longit_l = if q1.abs() > 1e-10 {
        (-cos_u * cos_v / q1).atan()
    } else {
        PI / 2.0
    };

    // Determine if we need 12-hour shift
    let is_best_am = if q1.abs() > 1e-10 {
        (-cos_u * cos_v / q1) > 0.0
    } else {
        true
    };
    let should_be_best_am = aspect > 0.0 && aspect <= PI;
    let time_offset = if should_be_best_am != is_best_am {
        PI
    } else {
        0.0
    };

    let latid_l = sin_phi_l.asin();
    let lum_c31_l = latid_l.cos() * cos_decl;
    let lum_c33_l = sin_phi_l * sin_decl;

    let s0 = lum_c31_l * (-time_angle - longit_l + time_offset).cos() + lum_c33_l;
    s0
}

//! Terrain derivatives: slope and aspect from DEM.
//!
//! Uses 3x3 finite difference method (Horn's method),
//! matching GRASS GIS r.slope.aspect.

use crate::types::Grid;

/// Compute slope and aspect from a DEM using Horn's method.
///
/// Returns (slope_grid, aspect_grid) where:
/// - slope is in radians (0 = flat, pi/2 = vertical)
/// - aspect is in degrees (GRASS convention: 0=east counterclockwise, then converted)
///   Output: 0=undefined (flat), degrees clockwise from east per GRASS internal convention
///
/// Edge pixels are set to nodata.
pub fn slope_aspect(dem: &Grid, ew_res: f64, ns_res: f64) -> (Grid, Grid) {
    let nodata = f32::NAN;
    let mut slope = Grid::new(dem.rows, dem.cols, nodata);
    let mut aspect = Grid::new(dem.rows, dem.cols, nodata);

    for r in 1..dem.rows - 1 {
        for c in 1..dem.cols - 1 {
            // Check for nodata in 3x3 neighborhood
            let mut has_nodata = false;
            for dr in 0..3 {
                for dc in 0..3 {
                    if dem.is_nodata(r + dr - 1, c + dc - 1) {
                        has_nodata = true;
                    }
                }
            }
            if has_nodata {
                continue;
            }

            // Horn's method (3x3 weighted differences)
            let z = |dr: i32, dc: i32| -> f64 {
                dem.get((r as i32 + dr) as usize, (c as i32 + dc) as usize) as f64
            };

            let dz_dx = ((z(1, 1) + 2.0 * z(0, 1) + z(-1, 1))
                - (z(1, -1) + 2.0 * z(0, -1) + z(-1, -1)))
                / (8.0 * ew_res);

            let dz_dy = ((z(-1, -1) + 2.0 * z(-1, 0) + z(-1, 1))
                - (z(1, -1) + 2.0 * z(1, 0) + z(1, 1)))
                / (8.0 * ns_res);

            let slope_rad = (dz_dx * dz_dx + dz_dy * dz_dy).sqrt().atan();
            slope.set(r, c, slope_rad as f32);

            if slope_rad.abs() < 1e-10 {
                aspect.set(r, c, 0.0); // flat — aspect undefined
            } else {
                // Compute aspect in GRASS cartographic convention:
                // 0/360 = north, 90 = east, 180 = south, 270 = west (clockwise from north)
                // This matches GRASS r.slope.aspect output.
                // atan2(dz_dx, -dz_dy) gives clockwise-from-north directly.
                let asp = dz_dx.atan2(-dz_dy).to_degrees();
                let asp = if asp < 0.0 { asp + 360.0 } else { asp };
                aspect.set(r, c, asp as f32);
            }
        }
    }

    (slope, aspect)
}

// rsun/rsun-core/src/horizon.rs
//! Horizon angle computation via ray-marching.
//!
//! For each pixel, march along N azimuthal directions and record
//! the maximum elevation angle to the horizon. This is the CPU
//! reference implementation; the GPU version uses compute shaders.

use crate::types::Grid;
use rayon::prelude::*;

/// Horizon angles for all pixels in all directions.
pub struct HorizonGrid {
    /// Stored as [rows * cols * n_directions], row-major
    pub angles: Vec<f32>,
    pub rows: usize,
    pub cols: usize,
    pub n_directions: usize,
}

impl HorizonGrid {
    /// Get horizon angles for a specific pixel.
    pub fn get_angles(&self, row: usize, col: usize) -> &[f32] {
        let offset = (row * self.cols + col) * self.n_directions;
        &self.angles[offset..offset + self.n_directions]
    }

    /// Interpolate horizon angle for a specific azimuth.
    pub fn interpolate(&self, row: usize, col: usize, azimuth: f64) -> f64 {
        let step = std::f64::consts::TAU / self.n_directions as f64;
        let pos = azimuth / step;
        let low = pos.floor() as usize % self.n_directions;
        let high = (low + 1) % self.n_directions;
        let frac = pos - pos.floor();
        let a_low = self.get_angles(row, col)[low] as f64;
        let a_high = self.get_angles(row, col)[high] as f64;
        a_low * (1.0 - frac) + a_high * frac
    }
}

/// Compute horizon angles for all pixels in the DEM.
///
/// Uses rayon for CPU parallelism over pixels.
///
/// # Arguments
/// * `dem` - Elevation grid
/// * `ew_res` - East-west cell resolution [meters]
/// * `ns_res` - North-south cell resolution [meters]
/// * `n_directions` - Number of azimuthal directions (e.g., 36 for 10-degree intervals)
pub fn compute_horizons(
    dem: &Grid,
    ew_res: f64,
    ns_res: f64,
    n_directions: usize,
) -> HorizonGrid {
    let step_dist = 0.5 * (ew_res + ns_res); // sampling distance
    let n_pixels = dem.rows * dem.cols;

    let angles: Vec<f32> = (0..n_pixels)
        .into_par_iter()
        .flat_map(|idx| {
            let row = idx / dem.cols;
            let col = idx % dem.cols;
            let mut pixel_angles = vec![0.0_f32; n_directions];

            if dem.is_nodata(row, col) {
                return pixel_angles;
            }

            let z0 = dem.get(row, col) as f64;
            let x0 = col as f64 * ew_res;
            let y0 = row as f64 * ns_res;

            for d in 0..n_directions {
                let azimuth = d as f64 * std::f64::consts::TAU / n_directions as f64;
                let dx = azimuth.cos() * step_dist;
                let dy = azimuth.sin() * step_dist;

                let mut max_angle: f64 = 0.0;
                let mut x = x0 + dx;
                let mut y = y0 + dy;

                loop {
                    // Convert to grid coordinates
                    let gc = (x / ew_res).round() as i64;
                    let gr = (y / ns_res).round() as i64;

                    // Bounds check
                    if gc < 0 || gc >= dem.cols as i64 || gr < 0 || gr >= dem.rows as i64 {
                        break;
                    }

                    let gr = gr as usize;
                    let gc = gc as usize;

                    if !dem.is_nodata(gr, gc) {
                        let z_remote = dem.get(gr, gc) as f64;
                        let dist = ((x - x0).powi(2) + (y - y0).powi(2)).sqrt();
                        if dist > 1e-6 {
                            let angle = ((z_remote - z0) / dist).atan();
                            if angle > max_angle {
                                max_angle = angle;
                            }
                        }
                    }

                    x += dx;
                    y += dy;
                }

                pixel_angles[d] = max_angle as f32;
            }

            pixel_angles
        })
        .collect();

    HorizonGrid {
        angles,
        rows: dem.rows,
        cols: dem.cols,
        n_directions,
    }
}

pub mod solar;

use types::{Grid, SolarParams, DayResult};
use horizon::HorizonGrid;

/// Compute solar radiation for a single day.
///
/// This is the core r.sun algorithm: loop from sunrise to sunset at `step`
/// intervals, computing beam + diffuse + reflected radiation at each time step,
/// accumulating into glob_rad and insol_time grids.
///
/// # Arguments
/// * `dem` - Elevation grid [meters]
/// * `slope` - Slope grid [radians]. Use all-zeros for flat.
/// * `aspect` - Aspect grid [degrees, GRASS convention]. Use all-zeros for flat.
/// * `lat_grid` - Latitude grid [radians]
/// * `lon_grid` - Longitude grid [radians]
/// * `horizons` - Optional pre-computed horizon angles
/// * `params` - Solar parameters (day, step, linke, albedo, solar_constant)
pub fn compute_day(
    dem: &Grid,
    slope: &Grid,
    aspect: &Grid,
    lat_grid: &Grid,
    lon_grid: &Grid,
    horizons: Option<&HorizonGrid>,
    params: &SolarParams,
) -> DayResult {
    let rows = dem.rows;
    let cols = dem.cols;
    let nodata = f32::NAN;

    let mut glob_rad = Grid::new(rows, cols, nodata);
    let mut insol_time = Grid::new(rows, cols, nodata);

    let decl = solar::declination(params.day);
    let g_norm_extra = solar::corrected_solar_constant(params.day, params.solar_constant);
    let step_hours = params.step;
    let step_rad = step_hours * solar::hourangle();

    use rayon::prelude::*;

    // Collect results in parallel, then write back
    let results: Vec<(usize, usize, f32, f32)> = (0..rows)
        .into_par_iter()
        .flat_map(|r| {
            let mut row_results = Vec::new();
            for c in 0..cols {
                if dem.is_nodata(r, c) {
                    continue;
                }

                let lat = if lat_grid.is_nodata(r, c) {
                    continue;
                } else {
                    lat_grid.get(r, c) as f64
                };

                let elevation = dem.get(r, c) as f64;
                let s = if slope.is_nodata(r, c) { 0.0 } else { slope.get(r, c) as f64 };
                let a = if aspect.is_nodata(r, c) { 0.0 } else { aspect.get(r, c) as f64 };

                // Aspect conversion: GRASS stores degrees, r.sun internally
                // converts to radians with 0=east CCW transformation
                let aspect_rad = a.to_radians();
                let aspect_internal = if a != 0.0 {
                    if a < 90.0 {
                        (90.0 - a).to_radians()
                    } else {
                        (450.0 - a).to_radians()
                    }
                } else {
                    0.0
                };

                let (sunrise, sunset) = solar::sunrise_sunset(lat, decl);

                // Determine first time step
                let sr_step_no = (sunrise / step_hours) as i32;
                let first_time = if (sunrise - sr_step_no as f64 * step_hours) > 0.5 * step_hours {
                    (sr_step_no as f64 + 1.5) * step_hours
                } else {
                    (sr_step_no as f64 + 0.5) * step_hours
                };

                let first_angle = solar::hour_to_time_angle(first_time);
                let last_angle = solar::hour_to_time_angle(sunset);

                let mut total_rad: f64 = 0.0;
                let mut total_insol: f64 = 0.0;
                let mut time_angle = first_angle;

                while time_angle <= last_angle {
                    let (solar_alt, solar_az) = solar::solar_position(lat, decl, time_angle);

                    if solar_alt > 0.0 {
                        // Shadow check
                        let is_shadowed = if let Some(h) = horizons {
                            let sun_az_angle = if solar_az < std::f64::consts::FRAC_PI_2 {
                                std::f64::consts::FRAC_PI_2 - solar_az
                            } else {
                                2.5 * std::f64::consts::PI - solar_az
                            };
                            let horizon_h = h.interpolate(r, c, sun_az_angle);
                            horizon_h > solar_alt
                        } else {
                            false // no shadow check without horizons
                        };

                        let s0 = radiation::cos_incidence(
                            s, aspect_internal, lat, decl, time_angle,
                        );

                        if !is_shadowed && s0 > 0.0 {
                            let (beam, beam_h) = radiation::brad(
                                s0, solar_alt, elevation, params.linke, 1.0, g_norm_extra,
                            );
                            total_rad += step_hours * beam;
                            total_insol += step_hours;

                            let (diff, refl) = radiation::drad(
                                s0, beam_h, solar_alt, solar_az,
                                s, aspect_internal, params.linke, params.albedo,
                                1.0, g_norm_extra,
                            );
                            total_rad += step_hours * (diff + refl);
                        } else {
                            // In shadow: still get diffuse
                            let (diff, refl) = radiation::drad(
                                s0.max(0.0), 0.0, solar_alt, solar_az,
                                s, aspect_internal, params.linke, params.albedo,
                                1.0, g_norm_extra,
                            );
                            total_rad += step_hours * (diff + refl);
                        }
                    }

                    time_angle += step_rad;
                }

                row_results.push((r, c, total_rad as f32, total_insol as f32));
            }
            row_results
        })
        .collect();

    for (r, c, rad, insol) in results {
        glob_rad.set(r, c, rad);
        insol_time.set(r, c, insol);
    }

    DayResult {
        day: params.day,
        glob_rad,
        insol_time,
    }
}

// rsun/rsun-core/src/io.rs
//! GeoTIFF I/O and coordinate transformation utilities.

use crate::types::{GeoTransform, Grid};
use gdal::raster::RasterBand;
use gdal::Dataset;

/// Read a GeoTIFF file into a Grid with geographic metadata.
pub fn read_geotiff(path: &str) -> Result<(Grid, GeoTransform), String> {
    let dataset = Dataset::open(path).map_err(|e| format!("Failed to open {path}: {e}"))?;

    let transform = dataset
        .geo_transform()
        .map_err(|e| format!("No geotransform: {e}"))?;

    let band: RasterBand = dataset
        .rasterband(1)
        .map_err(|e| format!("No raster band: {e}"))?;

    let (cols, rows) = dataset.raster_size();
    let nodata = band.no_data_value().unwrap_or(f64::NAN) as f32;

    let buf = band
        .read_as::<f32>((0, 0), (cols, rows), (cols, rows), None)
        .map_err(|e| format!("Failed to read raster data: {e}"))?;

    let mut grid = Grid::new(rows, cols, nodata);
    grid.data = buf.data().to_vec();

    let crs_wkt = dataset.projection();

    let geo = GeoTransform {
        x_origin: transform[0],
        y_origin: transform[3],
        x_res: transform[1].abs(),
        y_res: transform[5].abs(),
        crs_wkt: crs_wkt.unwrap_or_default(),
    };

    Ok((grid, geo))
}

/// Compute latitude/longitude grids from a projected DEM using the proj crate.
///
/// For each pixel, transforms the projected coordinate to geographic (lat/lon).
/// Returns (lat_grid, lon_grid) in radians.
pub fn compute_latlon_grid(
    rows: usize,
    cols: usize,
    geo: &GeoTransform,
) -> Result<(Grid, Grid), String> {
    use proj::Proj;

    let mut lat_grid = Grid::new(rows, cols, f32::NAN);
    let mut lon_grid = Grid::new(rows, cols, f32::NAN);

    // If already in lat/lon (EPSG:4326 or similar), just compute directly
    let is_geographic = geo.crs_wkt.contains("GEOGCS") && !geo.crs_wkt.contains("PROJCS");

    if is_geographic {
        for r in 0..rows {
            for c in 0..cols {
                let lon = geo.x_origin + (c as f64 + 0.5) * geo.x_res;
                let lat = geo.y_origin - (r as f64 + 0.5) * geo.y_res;
                lat_grid.set(r, c, lat.to_radians() as f32);
                lon_grid.set(r, c, lon.to_radians() as f32);
            }
        }
        return Ok((lat_grid, lon_grid));
    }

    // Transform from projected CRS to EPSG:4326
    let proj = Proj::new_known_crs(&geo.crs_wkt, "EPSG:4326", None)
        .map_err(|e| format!("Failed to create projection: {e}"))?;

    for r in 0..rows {
        for c in 0..cols {
            let x = geo.x_origin + (c as f64 + 0.5) * geo.x_res;
            let y = geo.y_origin - (r as f64 + 0.5) * geo.y_res;
            match proj.convert((x, y)) {
                Ok((lon, lat)) => {
                    lat_grid.set(r, c, lat.to_radians() as f32);
                    lon_grid.set(r, c, lon.to_radians() as f32);
                }
                Err(_) => {} // leave as NaN
            }
        }
    }

    Ok((lat_grid, lon_grid))
}

/// Write a Grid to a GeoTIFF file.
pub fn write_geotiff(
    path: &str,
    grid: &Grid,
    geo: &GeoTransform,
    source_path: Option<&str>,
) -> Result<(), String> {
    use gdal::DriverManager;

    let driver = DriverManager::get_driver_by_name("GTiff")
        .map_err(|e| format!("GTiff driver not found: {e}"))?;

    let mut dataset = driver
        .create_with_band_type::<f32, &str>(
            path,
            grid.cols as isize,
            grid.rows as isize,
            1,
        )
        .map_err(|e| format!("Failed to create output: {e}"))?;

    // Set geotransform
    let transform = [geo.x_origin, geo.x_res, 0.0, geo.y_origin, 0.0, -geo.y_res];
    dataset
        .set_geo_transform(&transform)
        .map_err(|e| format!("Failed to set geotransform: {e}"))?;

    // Copy projection from source if available
    if let Some(src_path) = source_path {
        if let Ok(src_ds) = Dataset::open(src_path) {
            if let Ok(proj) = src_ds.projection() {
                let _ = dataset.set_projection(&proj);
            }
        }
    }

    // Write data
    let mut band = dataset
        .rasterband(1)
        .map_err(|e| format!("Failed to get band: {e}"))?;

    band.set_no_data_value(Some(grid.nodata as f64))
        .map_err(|e| format!("Failed to set nodata: {e}"))?;

    let buf = gdal::raster::Buffer::new((grid.cols, grid.rows), grid.data.clone());
    band.write((0, 0), (grid.cols, grid.rows), &buf)
        .map_err(|e| format!("Failed to write data: {e}"))?;

    Ok(())
}

use rsun_core::{compute_day, io, solar, terrain, types::SolarParams};

#[derive(Parser)]
#[command(name = "rsun", about = "GPU-accelerated solar radiation modeling")]
struct Cli {
    #[command(subcommand)]
    command: Commands,
}

#[derive(Subcommand)]
enum Commands {
    /// Compute solar radiation
    Compute {
        /// Input DEM GeoTIFF
        #[arg(long)]
        dem: String,

        /// Day of year (1-366) or range (e.g., "1-365")
        #[arg(long)]
        day: String,

        /// Time step in decimal hours
        #[arg(long, default_value = "0.5")]
        step: f64,

        /// Linke turbidity factor
        #[arg(long, default_value = "3.0")]
        linke: f64,

        /// Surface albedo
        #[arg(long, default_value = "0.2")]
        albedo: f64,

        /// Calendar year (enables leap year: 366 days when "all" is used)
        #[arg(long)]
        year: Option<u16>,

        /// Output global radiation GeoTIFF (single day mode)
        #[arg(long)]
        glob_rad: Option<String>,

        /// Output insolation time GeoTIFF (single day mode)
        #[arg(long)]
        insol_time: Option<String>,

        /// Output directory (multi-day mode)
        #[arg(long)]
        output_dir: Option<String>,
    },
    /// Compute slope and aspect from DEM

        #[arg(long)]
        dem: String,
        #[arg(long)]
        slope_out: String,
        #[arg(long)]
        aspect_out: String,
    },
    /// Show system info
    Info,
}

fn main() {
    let cli = Cli::parse();

    match cli.command {
        Commands::Compute {
            dem,
            day,
            step,
            linke,
            albedo,
            glob_rad,
            insol_time,
            output_dir,
        } => {
            eprintln!("Reading DEM: {dem}");
            let (dem_grid, geo) = io::read_geotiff(&dem).expect("Failed to read DEM");
            eprintln!("DEM size: {} rows x {} cols", dem_grid.rows, dem_grid.cols);

            // Compute slope and aspect
            eprintln!("Computing slope and aspect...");
            let (slope, aspect) = terrain::slope_aspect(&dem_grid, geo.x_res, geo.y_res);

            // Compute lat/lon grids from DEM CRS via proj crate
            eprintln!("Computing lat/lon grid from CRS...");
            let (lat_grid, lon_grid) = io::compute_latlon_grid(
                dem_grid.rows, dem_grid.cols, &geo,
            ).expect("Failed to compute lat/lon grid from CRS");

            // Parse day specification
            let days = parse_days(&day, year);

            for d in &days {
                let params = SolarParams {
                    day: *d,
                    step,
                    linke,
                    albedo,
                    solar_constant: 1367.0,
                };

                eprintln!("Computing day {}...", d);
                let result = compute_day(
                    &dem_grid, &slope, &aspect,
                    &lat_grid, &lon_grid,
                    None, &params,
                );

                // Write output
                if days.len() == 1 {
                    if let Some(ref path) = glob_rad {
                        io::write_geotiff(path, &result.glob_rad, &geo, Some(&dem))
                            .expect("Failed to write glob_rad");
                        eprintln!("Wrote: {path}");
                    }
                    if let Some(ref path) = insol_time {
                        io::write_geotiff(path, &result.insol_time, &geo, Some(&dem))
                            .expect("Failed to write insol_time");
                        eprintln!("Wrote: {path}");
                    }
                } else if let Some(ref dir) = output_dir {
                    std::fs::create_dir_all(format!("{dir}/global/daily")).ok();
                    std::fs::create_dir_all(format!("{dir}/insol/daily")).ok();
                    let grad_path = format!("{dir}/global/daily/total_sun_day_{d}.tif");
                    let insol_path = format!("{dir}/insol/daily/hours_sun_day_{d}.tif");
                    io::write_geotiff(&grad_path, &result.glob_rad, &geo, Some(&dem))
                        .expect("Failed to write glob_rad");
                    io::write_geotiff(&insol_path, &result.insol_time, &geo, Some(&dem))
                        .expect("Failed to write insol_time");
                }
            }
            eprintln!("Done. Processed {} days.", days.len());
        }
        Commands::Terrain { dem, slope_out, aspect_out } => {

        Commands::Info => {

            println!("Backend: CPU (rayon)");
            println!("GPU: not yet implemented (Phase 2)");
        }
    }
}

fn parse_days(spec: &str, year: Option<u16>) -> Vec<u16> {
    let max_day = match year {
        Some(y) if is_leap_year(y) => 366,
        _ => 365,
    };
    if spec == "all" {
        return (1..=max_day).collect();
    }
    if let Some((start, end)) = spec.split_once('-') {
        let s: u16 = start.parse().expect("Invalid day range start");
        let e: u16 = end.parse().expect("Invalid day range end");
        return (s..=e.min(max_day)).collect();
    }
    vec![spec.parse().expect("Invalid day number")]
}

fn is_leap_year(year: u16) -> bool {
    (year % 4 == 0 && year % 100 != 0) || (year % 400 == 0)
}
